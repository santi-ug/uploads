(() => {
  'use strict';
  const host = document.createElement('div');
  host.style.cssText = 'position:fixed;inset:0;z-index:2147483647;pointer-events:none';
  const shadow = host.attachShadow({ mode: 'closed' }), style = document.createElement('style');
  style.textContent = ':host{color-scheme:light dark}svg{position:absolute;inset:0;width:100%;height:100%;overflow:hidden}button{position:absolute;pointer-events:auto;min-height:44px;padding:8px 14px;border:0;border-radius:8px;background:#171717;color:white;font:14px system-ui;box-shadow:0 0 0 1px #fff5,0 4px 18px #0003;cursor:pointer}button:focus-visible{outline:3px solid #85adff;outline-offset:3px}[hidden]{display:none!important}@media(prefers-color-scheme:dark){button{background:white;color:black}}';
  const ns = 'http://www.w3.org/2000/svg', svg = document.createElementNS(ns, 'svg'); svg.setAttribute('aria-hidden', 'true');
  const action = document.createElement('button'); action.textContent = 'Comment'; action.hidden = true;
  const savedSvg = document.createElementNS(ns, 'svg'); savedSvg.setAttribute('aria-hidden', 'true');
  const pins = document.createElement('div');
  style.textContent += '.pin{width:28px;height:28px;min-height:28px;padding:0;border:2px solid white;border-radius:50% 50% 50% 4px;background:var(--pin-color);color:white;font:600 10px system-ui;box-shadow:0 1px 4px #0004}.pin[aria-expanded=true]{outline:2px solid #2563eb;outline-offset:2px}.pin:active{transform:scale(.95)}@media(pointer:coarse){.pin:before{content:"";position:absolute;inset:-8px}}';
  shadow.append(style, savedSvg, svg, pins, action); document.documentElement.append(host);
  let tool = 'read', allowed = false, replay = false, marks = [], undo = [], redo = [], viewport = null;
  let shortcuts = [], threads = [], selectedThread = null, threadKey = '';
  const pinNodes = new Map();
  let highlighted = null;
  let active = null, pending = null, quote = '', selector = '', frame = 0, timer;
  const clean = value => (value || '').replace(/\s+/g, ' ').trim().slice(0, 1000);
  const send = value => parent.postMessage({ ...value, revision: reviewRevision }, '*');
  const point = event => ({ x: Math.max(0, Math.min(200000, Math.round((event.clientX + scrollX) * 10) / 10)), y: Math.max(0, Math.min(200000, Math.round((event.clientY + scrollY) * 10) / 10)) });
  function pathFor(element) {
    const parts = [];
    while (element && element !== document.body && parts.length < 12) {
      const tag = element.localName, peers = [...element.parentElement?.children || []].filter(node => node.localName === tag);
      parts.unshift(tag + ':nth-of-type(' + (peers.indexOf(element) + 1) + ')'); element = element.parentElement;
    }
    return parts.join(' > ').slice(0, 500);
  }
  function bounds(mark) {
    if (mark.kind !== 'stroke') return mark;
    const xs = mark.points.map(p => p.x), ys = mark.points.map(p => p.y);
    return { x: Math.min(...xs), y: Math.min(...ys), width: Math.max(...xs) - Math.min(...xs), height: Math.max(...ys) - Math.min(...ys) };
  }
  const rect = r => ({ kind: 'rect', x: Math.max(0, r.left + scrollX), y: Math.max(0, r.top + scrollY), width: Math.max(1, r.width), height: Math.max(1, r.height) });
  const payload = () => marks.length ? { version: 1, viewport, marks } : null;
  const state = () => ({ marks, viewport, quote, selector });
  function restore(value) { ({ marks, viewport, quote, selector } = value); }
  function remember() { undo.push(state()); if (undo.length > 64) undo.shift(); redo = []; }
  function changed() {
    if (!marks.length) { viewport = null; quote = ''; selector = ''; }
    send({ type: 'review-markup', markup: payload(), quote, selector, canUndo: undo.length > 0, canRedo: redo.length > 0 }); paint();
  }
  function append(next, anchor) {
    if (marks.length + next.length > 32 || [...marks, ...next].reduce((n, m) => n + (m.points?.length || 0), 0) > 2048) {
      send({ type: 'review-notice', text: 'Annotation limit reached. Post this note before adding more marks.' }); return false;
    }
    remember(); if (anchor) ({ quote, selector } = anchor);
    viewport ||= { width: innerWidth, height: innerHeight }; marks = [...marks, ...next]; changed(); return true;
  }
  function drawMark(mark, layer, opacity) {
    let node;
    if (mark.kind === 'stroke') {
      node = document.createElementNS(ns, 'polyline'); node.setAttribute('points', mark.points.map(p => `${p.x - scrollX},${p.y - scrollY}`).join(' '));
    } else {
      node = document.createElementNS(ns, mark.kind === 'ellipse' ? 'ellipse' : 'rect');
      const attributes = mark.kind === 'ellipse' ? { cx: mark.x - scrollX + mark.width / 2, cy: mark.y - scrollY + mark.height / 2, rx: mark.width / 2, ry: mark.height / 2 } : { x: mark.x - scrollX, y: mark.y - scrollY, width: mark.width, height: mark.height };
      for (const [key, value] of Object.entries(attributes)) node.setAttribute(key, value);
    }
    node.setAttribute('fill', 'none'); node.setAttribute('stroke', '#2563eb'); node.setAttribute('stroke-width', '3'); node.setAttribute('stroke-linecap', 'round'); node.setAttribute('stroke-linejoin', 'round'); node.setAttribute('opacity', opacity); layer.append(node);
  }
  // Saved annotations are independent of the editable draft and its undo history.
  function threadGeometry(thread) {
    const source = thread.markup?.marks || [];
    const scale = thread.markup ? innerWidth / thread.markup.viewport.width : 1;
    let mapped = source.map(mark => mark.kind === 'stroke'
      ? { ...mark, points: mark.points.map(p => ({ x: p.x * scale, y: p.y * scale })) }
      : { ...mark, x: mark.x * scale, y: mark.y * scale, width: mark.width * scale, height: mark.height * scale });
    let target = null;
    if (thread.selector) {
      try {
        const node = document.querySelector(thread.selector);
        if (node && clean(node.textContent).includes(clean(thread.quote))) target = rect(node.getBoundingClientRect());
      } catch { /* A missing anchor falls back to saved geometry. */ }
    }
    if (target && mapped.length && scale !== 1) {
      const first = bounds(mapped[0]), dx = target.x - first.x, dy = target.y - first.y;
      mapped = mapped.map(mark => mark.kind === 'stroke'
        ? { ...mark, points: mark.points.map(p => ({ x: p.x + dx, y: p.y + dy })) }
        : { ...mark, x: mark.x + dx, y: mark.y + dy });
    }
    const first = mapped.length ? bounds(mapped[0]) : target;
    return { marks: mapped, anchor: first ? { x: Math.max(0, first.x + first.width - 14), y: Math.max(0, first.y - 14) } : null };
  }
  function focusThread(id) {
    const thread = threads.find(item => item.id === id);
    if (!thread) return;
    selectedThread = id;
    const geometry = threadGeometry(thread);
    if (geometry.anchor) scrollTo({ top: Math.max(0, geometry.anchor.y - 120), left: Math.max(0, geometry.anchor.x - innerWidth + 100), behavior: 'instant' });
    paint();
  }
  function paint() {
    if (frame) return;
    frame = requestAnimationFrame(() => {
      frame = 0; svg.replaceChildren();
      savedSvg.replaceChildren();
      for (const thread of threads) {
        const geometry = threadGeometry(thread);
        for (const mark of geometry.marks) drawMark(mark, savedSvg, thread.id === selectedThread ? '1' : '.5');
        const pin = pinNodes.get(thread.id);
        pin.hidden = !geometry.anchor;
        pin.setAttribute('aria-expanded', String(thread.id === selectedThread));
        if (geometry.anchor) {
          const x = geometry.anchor.x - scrollX, y = geometry.anchor.y - scrollY;
          pin.style.left = x + 'px'; pin.style.top = y + 'px';
          if (thread.id === selectedThread) send({ type: 'review-thread-position', id: thread.id, position: { x: x + 36, y } });
        }
      }
      for (const mark of [...marks, ...(highlighted ? [highlighted] : []), ...(active ? [active.mark] : [])]) drawMark(mark, svg, '1');
      action.hidden = replay || !allowed || active !== null || (!marks.length && !pending);
      if (!action.hidden) {
        const b = pending?.rect || bounds(marks[marks.length - 1]);
        action.style.left = Math.max(8, Math.min(innerWidth - 105, b.x - scrollX + b.width - 90)) + 'px';
        action.style.top = Math.max(8, Math.min(innerHeight - 52, b.y - scrollY + b.height + 8)) + 'px';
      }
    });
  }
  function setTool(next) {
    if (next !== 'read') highlighted = null;
    tool = allowed && !replay && ['select', 'rect', 'ellipse', 'stroke', 'erase'].includes(next) ? next : 'read';
    host.style.pointerEvents = tool === 'read' ? 'none' : 'auto'; host.style.touchAction = tool === 'read' ? 'auto' : 'none';
    host.style.cursor = tool === 'read' ? '' : tool === 'erase' ? 'not-allowed' : 'crosshair'; pending = null; active = null; paint();
  }
  function ready() { send({ type: 'review-ready', title: document.title.slice(0, 200) }); }
  addEventListener('message', event => {
    if (event.source !== parent || !event.data || typeof event.data !== 'object') return;
    const message = event.data;
    if (message.type === 'review-threads' && message.revision === reviewRevision && Array.isArray(message.threads)) {
      const key = JSON.stringify(message.threads);
      if (key !== threadKey) {
        threadKey = key; threads = message.threads; pins.replaceChildren(); pinNodes.clear();
        for (const thread of threads) {
          const pin = document.createElement('button'); pin.className = 'pin';
          pin.textContent = thread.initial; pin.style.setProperty('--pin-color', thread.color);
          pin.setAttribute('aria-label', 'Comment by ' + thread.name + ': ' + thread.body);
          pin.addEventListener('pointerdown', event => event.stopPropagation());
          pin.addEventListener('click', () => { focusThread(thread.id); send({ type: 'review-open-thread', id: thread.id }); });
          pinNodes.set(thread.id, pin); pins.append(pin);
        }
      }
      selectedThread = message.selectedId; paint();
    }
    if (message.type === 'review-focus-thread' && message.revision === reviewRevision) focusThread(message.id);
    if (message.type === 'review-ping') ready();
    if (message.type === 'review-tool') { allowed = message.allowed === true; shortcuts = Array.isArray(message.shortcuts) ? message.shortcuts.filter(key => typeof key === 'string') : []; setTool(message.tool); }
    if (message.type === 'review-history' && !replay) {
      if (message.action === 'undo' && undo.length) { redo.push(state()); restore(undo.pop()); }
      if (message.action === 'redo' && redo.length) { undo.push(state()); restore(redo.pop()); }
      changed();
    }
    if (message.type === 'review-restore' && message.revision === reviewRevision) {
      marks = message.markup?.marks || []; viewport = message.markup?.viewport || null; quote = clean(message.quote); selector = typeof message.selector === 'string' ? message.selector.slice(0, 500) : '';
      undo = marks.length ? [{ marks: [], viewport: null, quote: '', selector: '' }] : []; redo = []; replay = message.replay === true; setTool('read'); paint();
      if (replay && marks.length) { const b = bounds(marks[0]); scrollTo({ top: Math.max(0, b.y - 100), left: Math.max(0, b.x - 100), behavior: 'instant' }); }
    }
    if (message.type === 'review-clear') { marks = []; undo = []; redo = []; pending = null; highlighted = null; replay = false; changed(); }
    if (message.type === 'review-reveal' && message.revision === reviewRevision && typeof message.selector === 'string') {
      try { const node = document.querySelector(message.selector); if (node && clean(node.textContent).includes(clean(message.quote))) { node.scrollIntoView({ block: 'center' }); highlighted = rect(node.getBoundingClientRect()); setTool('read'); paint(); } } catch { /* Quotes remain readable when a selector no longer exists. */ }
    }
  });
  const ownControl = event => {
    if (event.composedPath().includes(action)) return true;
    for (const pin of pinNodes.values()) {
      const r = pin.getBoundingClientRect();
      if (!pin.hidden && event.clientX >= r.left - 8 && event.clientX <= r.right + 8 && event.clientY >= r.top - 8 && event.clientY <= r.bottom + 8) return true;
    }
    // Closed-shadow events are retargeted to the host at document capture time.
    const r = action.getBoundingClientRect();
    return !action.hidden && event.clientX >= r.left && event.clientX <= r.right && event.clientY >= r.top && event.clientY <= r.bottom;
  };
  document.addEventListener('pointerdown', event => {
    if (tool === 'read' || !allowed || event.button !== 0 || ownControl(event)) return;
    // preventDefault suppresses the browser's normal click-into-iframe focus
    // transfer, so a keyboard Enter right after drawing would otherwise land
    // on the parent page instead of here.
    event.preventDefault(); event.stopImmediatePropagation(); focus(); const p = point(event);
    if (tool === 'erase') {
      const index = marks.findLastIndex(mark => { const b = bounds(mark); return p.x >= b.x - 8 && p.x <= b.x + b.width + 8 && p.y >= b.y - 8 && p.y <= b.y + b.height + 8; });
      if (index >= 0) { remember(); marks = marks.filter((_, i) => i !== index); changed(); } return;
    }
    if (tool === 'select') {
      const node = document.elementsFromPoint(event.clientX, event.clientY).find(node => node !== host && node !== document.body && node !== document.documentElement);
      if (!node) return;
      const target = node.closest('img,svg,canvas,video,p,h1,h2,h3,h4,li,td,th,button,a,figure') || node;
      append([rect(target.getBoundingClientRect())], { quote: clean(target.textContent || target.getAttribute('alt') || target.getAttribute('aria-label') || target.localName), selector: pathFor(target) }); return;
    }
    active = { id: event.pointerId, start: p, mark: tool === 'stroke' ? { kind: 'stroke', points: [p] } : { kind: tool, ...p, width: 1, height: 1 } };
    host.setPointerCapture(event.pointerId); paint();
  }, true);
  document.addEventListener('pointermove', event => {
    if (!active || active.id !== event.pointerId) return;
    event.preventDefault(); const p = point(event), mark = active.mark;
    if (mark.kind === 'stroke') {
      const previous = mark.points[mark.points.length - 1]; if (Math.hypot(p.x - previous.x, p.y - previous.y) >= 2 && mark.points.length < 512) mark.points.push(p);
    } else { mark.x = Math.min(active.start.x, p.x); mark.y = Math.min(active.start.y, p.y); mark.width = Math.max(1, Math.abs(p.x - active.start.x)); mark.height = Math.max(1, Math.abs(p.y - active.start.y)); }
    paint();
  }, { capture: true, passive: false });
  function finish(event) {
    if (!active || active.id !== event.pointerId) return;
    const mark = active.mark; active = null;
    if (event.type !== 'pointercancel' && (mark.kind === 'stroke' ? mark.points.length > 1 : mark.width > 2 && mark.height > 2)) append([mark]);
    if (host.hasPointerCapture(event.pointerId)) host.releasePointerCapture(event.pointerId); paint();
  }
  document.addEventListener('pointerup', finish, true); document.addEventListener('pointercancel', finish, true);
  document.addEventListener('click', event => { if (tool !== 'read' && !ownControl(event)) { event.preventDefault(); event.stopImmediatePropagation(); } }, true);
  document.addEventListener('selectionchange', () => {
    clearTimeout(timer); timer = setTimeout(() => {
      if (tool !== 'read' || !allowed || replay) return;
      const selection = getSelection(); if (!selection || selection.isCollapsed || !clean(selection.toString())) { pending = null; paint(); return; }
      const range = selection.getRangeAt(0), node = selection.anchorNode?.parentElement; if (!node || node === host) return;
      pending = { quote: clean(selection.toString()), selector: pathFor(node), rect: rect(range.getBoundingClientRect()), marks: [...range.getClientRects()].filter(r => r.width > 0 && r.height > 0).slice(0, 32).map(rect) }; paint();
    }, 200);
  });
  action.addEventListener('pointerdown', event => event.preventDefault());
  function openComment() {
    if (pending) { if (!append(pending.marks, pending)) return; pending = null; }
    const b = action.getBoundingClientRect(); send({ type: 'review-target', quote, selector, markup: payload(), position: { x: b.left, y: b.bottom } });
  }
  action.addEventListener('click', openComment);
  document.addEventListener('keydown', event => {
    if (event.key === 'Escape') send({ type: 'review-escape' });
    // Same visibility condition paint() uses for the floating Comment button
    // (that value updates on the next animation frame, so it lags a fresh mark).
    if (event.key === 'Enter' && ![...pinNodes.values()].includes(shadow.activeElement) && !event.repeat && allowed && !replay && active === null && (marks.length || pending)) {
      const editable = event.composedPath().some(node => node.isContentEditable || node.matches?.('input,textarea,select,[role="textbox"],[role="combobox"]'));
      if (!editable) { event.preventDefault(); openComment(); }
    }
    const key = reviewShortcut(event);
    if (allowed && !replay && shortcuts.includes(key)) { event.preventDefault(); send({ type: 'review-shortcut', key }); }
  });
  addEventListener('scroll', paint, { passive: true }); addEventListener('resize', paint); ready();
})();
