(() => {
  'use strict';
  const byId = id => document.getElementById(id);
  const frame = byId('document'), panel = byId('panel'), form = byId('composer');
  const name = byId('name'), body = byId('body'), status = byId('status');
  const slug = document.body.dataset.slug;
  const endpoint = '/' + slug + '/comments';
  const draftKey = 'review-draft:' + slug;
  const nameKey = 'review-name';
  // Tailnet HTTP previews lack randomUUID; getRandomValues works on HTTP too.
  function newRequestId() {
    if (typeof crypto.randomUUID === 'function') return crypto.randomUUID();
    const bytes = crypto.getRandomValues(new Uint8Array(16));
    bytes[6] = (bytes[6] & 15) | 64;
    bytes[8] = (bytes[8] & 63) | 128;
    const hex = Array.from(bytes, byte => byte.toString(16).padStart(2, '0')).join('');
    return [hex.slice(0, 8), hex.slice(8, 12), hex.slice(12, 16), hex.slice(16, 20), hex.slice(20)].join('-');
  }
  let data = { revision: '', open: false, comments: [] };
  let frameRevision = '', busy = false, loading = false, tool = 'read', viewing = null, loadingSnapshot = false;
  let notifiedCount = 0;
  let markup = null;
  const popover = byId('compose-popover'), threadPopover = byId('thread-popover');
  let selectedId = null, threadPosition = null, threadContentKey = '', threadListKey = '', threadReturnFocus = null;
  let anchor = { quote: '', selector: '' }, parentId = null, requestId = newRequestId();
  let returnFocus = byId('show');
  function saveDraft() {
    try {
      localStorage.setItem(nameKey, name.value);
      localStorage.setItem(draftKey, JSON.stringify({ body: body.value, anchor, markup, parentId, requestId, revision: draftRevision || frameRevision }));
    } catch { /* Posting works without browser storage. */ }
    byId('draft').hidden = !body.value || !popover.hidden || Boolean(viewing);
  }
  // The name is remembered globally (not per document) so it survives a new writeup's new slug.
  try {
    const savedName = localStorage.getItem(nameKey);
    if (typeof savedName === 'string') name.value = savedName.slice(0, 60);
  } catch { /* A missing or invalid saved name never prevents reading. */ }
  let draftRevision = '';
  try {
    const draft = JSON.parse(localStorage.getItem(draftKey) || 'null');
    if (draft && typeof draft.body === 'string') {
      body.value = draft.body.slice(0, 2000);
      if (draft.anchor && typeof draft.anchor.quote === 'string' && typeof draft.anchor.selector === 'string') anchor = { quote: draft.anchor.quote.slice(0, 1000), selector: draft.anchor.selector.slice(0, 500) };
      if (typeof draft.parentId === 'string') parentId = draft.parentId;
      if (typeof draft.requestId === 'string') requestId = draft.requestId;
      if (typeof draft.revision === 'string') draftRevision = draft.revision;
      markup = safeMarkup(draft.markup);
    }
  } catch { /* A missing or invalid draft never prevents reading. */ }
  function openPanel(focus = false) {
    closeThread(false);
    if (!popover.hidden) closeComposer();
    if (panel.hidden) {
      const active = document.activeElement;
      returnFocus = active instanceof HTMLElement && active !== document.body && !panel.contains(active) ? active : byId('show');
    }
    panel.hidden = false; document.body.classList.add('panel-open'); byId('show').setAttribute('aria-expanded', 'true');
    if (focus) byId('close').focus();
  }
  function closePanel() {
    // Desktop keeps the panel open permanently (Figma/Google-Docs style, no hidden inbox);
    // only the mobile bottom sheet actually closes.
    if (!matchMedia('(max-width:760px)').matches) return;
    panel.hidden = true; document.body.classList.remove('panel-open'); byId('show').setAttribute('aria-expanded', 'false');
    returnFocus.focus();
  }
  function safeMarkup(value) {
    if (!value || value.version !== 1 || !value.viewport || !Array.isArray(value.marks) || !value.marks.length || value.marks.length > 32) return null;
    const finite = (n, min, max) => typeof n === 'number' && Number.isFinite(n) && n >= min && n <= max;
    if (!finite(value.viewport.width, 240, 8000) || !finite(value.viewport.height, 160, 8000)) return null;
    let points = 0;
    for (const m of value.marks) {
      if (!m || typeof m !== 'object') return null;
      if (m.kind === 'stroke') {
        if (!Array.isArray(m.points) || m.points.length < 2 || m.points.length > 512) return null;
        points += m.points.length;
        if (points > 2048 || !m.points.every(p => p && finite(p.x, 0, 200000) && finite(p.y, 0, 200000))) return null;
      } else if (!['rect', 'ellipse'].includes(m.kind) || !finite(m.x, 0, 200000) || !finite(m.y, 0, 200000) || !finite(m.width, 1, 20000) || !finite(m.height, 1, 20000)) return null;
    }
    return value;
  }
  const tell = message => frame.contentWindow?.postMessage(message, '*');
  function clearMarks() {
    markup = null; tell({ type: 'review-clear' }); sizeFrame(null);
    for (const action of ['undo', 'redo']) document.querySelector('[data-tool="' + action + '"]').disabled = true;
  }
  function sizeFrame(value) {
    frame.style.width = value ? value.viewport.width + 'px' : '';
    frame.style.height = value ? value.viewport.height + 'px' : '';
    frame.style.minWidth = value ? value.viewport.width + 'px' : '';
  }
  function setTool(next) {
    tool = data.open && !viewing && !busy ? next : 'read';
    for (const button of document.querySelectorAll('[data-tool][aria-pressed]')) button.setAttribute('aria-pressed', String(button.dataset.tool === tool));
    tell({ type: 'review-tool', tool, allowed: data.open && !viewing && !busy, shortcuts: [...document.querySelectorAll('[data-shortcut]')].map(button => button.dataset.shortcut) });
    const hints = { read: 'Select text to comment. Read also lets you scroll.', select: 'Tap an element, then Comment beside it.', rect: 'Drag a region, then Comment beside it.', ellipse: 'Drag to circle anything, including images.', stroke: 'Draw over the page, then Comment beside the marks.', erase: 'Tap a mark to remove it. Undo restores it.' };
    byId('instruction').textContent = hints[tool];
  }
  function closeComposer() { popover.hidden = true; saveDraft(); returnFocus.focus(); }
  function openComposer(position) {
    closeThread(false);
    popover.hidden = false; byId('draft').hidden = true;
    const r = frame.getBoundingClientRect();
    popover.style.left = Math.max(12, Math.min(innerWidth - 372, r.left + (position?.x || 24))) + 'px';
    popover.style.top = Math.max(70, Math.min(innerHeight - popover.offsetHeight - 96, r.top + (position?.y || 60))) + 'px';
    if (!matchMedia('(pointer:coarse)').matches) body.focus({ preventScroll: true });
  }
  function targetLabel() {
    byId('composer-title').textContent = parentId ? 'Reply' : 'New comment';
    byId('target').textContent = anchor.quote || (markup ? markup.marks.length + ' visual marks' : '');
    byId('target').hidden = !anchor.quote && !markup;
    byId('clear-target').hidden = !anchor.quote && !parentId && !markup;
  }
  function selectTarget(next, thread = null, position, nextMarkup = markup) {
    if (busy) return;
    if (body.value.trim() && (thread !== parentId || anchor.selector !== next.selector || anchor.quote !== next.quote) && !confirm('Move your current draft to this comment target?')) { tell({ type: 'review-restore', markup, ...anchor, revision: frameRevision }); return; }
    anchor = next; markup = nextMarkup; parentId = thread; requestId = newRequestId(); draftRevision = frameRevision;
    if (!nextMarkup) clearMarks();
    targetLabel(); saveDraft(); if (!panel.hidden) closePanel(); openComposer(position); setTool('read');
  }
  const element = (tag, text, className) => {
    const node = document.createElement(tag); node.textContent = text;
    if (className) node.className = className; return node;
  };
  const button = (label, action) => {
    const node = element('button', label); node.type = 'button'; node.addEventListener('click', action); return node;
  };
  // Deterministic per-name color so the same commenter always gets the same
  // avatar, without tracking identity beyond the self-reported name.
  function avatarColor(value) {
    let hash = 0; for (let i = 0; i < value.length; i++) hash = (hash * 31 + value.charCodeAt(i)) >>> 0;
    return 'hsl(' + (hash % 360) + ' 55% 45%)';
  }
  function commentHead(commenter, when, suffix) {
    const head = element('div', '', 'comment-head');
    const dot = element('span', ([...commenter.trim()][0] || '?').toUpperCase(), 'avatar');
    dot.style.background = avatarColor(commenter);
    head.append(dot, element('span', commenter + ' · ' + when + suffix, 'meta'));
    return head;
  }
  // A native notification when the open count grows, so feedback surfaces without
  // requiring a chat message back to whoever is waiting on it. Same-origin, no new
  // server capability: the comment count was already public and already fetched.
  function notify(openCount) {
    if (typeof Notification === 'undefined' || openCount <= notifiedCount) { notifiedCount = openCount; return; }
    if (Notification.permission === 'default') Notification.requestPermission();
    if (Notification.permission === 'granted') new Notification('New review feedback', { body: openCount + ' open comment' + (openCount === 1 ? '' : 's') + ' on ' + (byId('title').textContent || 'this document') });
    notifiedCount = openCount;
  }
  function render() {
    name.disabled = busy; body.disabled = busy;
    byId('review-state').textContent = data.open ? '' : 'Review closed';
    byId('post').disabled = busy || !data.open || !frameRevision || Boolean(viewing) || (!parentId && Boolean(draftRevision) && draftRevision !== frameRevision);
    for (const button of document.querySelectorAll('[data-tool]:not([data-tool="read"]):not([data-tool="undo"]):not([data-tool="redo"])')) button.disabled = !data.open || Boolean(viewing) || busy;
    byId('new').disabled = !data.open || Boolean(viewing) || busy;
    if (!data.open || viewing || busy) for (const action of ['undo', 'redo']) document.querySelector('[data-tool="' + action + '"]').disabled = true;
    if (!data.open) setTool('read');
    const roots = data.comments.filter(comment => comment.parent_id === null);
    const openCount = roots.filter(comment => comment.status === 'open').length;
    byId('count').textContent = openCount;
    notify(openCount);
    const list = byId('threads');
    const visible = roots.filter(comment => byId('resolved').checked || comment.status === 'open');
    const listKey = JSON.stringify([visible, data.comments.filter(comment => comment.parent_id !== null), frameRevision]);
    if (listKey !== threadListKey) {
      threadListKey = listKey; list.replaceChildren();
      for (const comment of visible) {
        const row = button('', () => openThread(comment.id)); row.className = 'thread-row'; row.dataset.thread = comment.id;
        row.setAttribute('aria-expanded', String(selectedId === comment.id)); row.setAttribute('aria-controls', 'thread-popover');
        row.append(commentHead(comment.name, new Date(comment.created_at).toLocaleDateString(undefined, { month: 'short', day: 'numeric' }), ''));
        row.append(element('span', comment.body, 'thread-excerpt'));
        const replies = data.comments.filter(item => item.parent_id === comment.id).length;
        const notes = [comment.revision !== frameRevision ? 'Earlier version' : '', comment.status === 'resolved' ? 'Resolved' : '', replies ? replies + (replies === 1 ? ' reply' : ' replies') : ''].filter(Boolean);
        if (notes.length) row.append(element('span', notes.join(' · '), 'meta'));
        list.append(row);
      }
      if (!list.childElementCount) list.append(element('p', 'No comments yet.', 'hint'));
    }
    if (selectedId && !visible.some(comment => comment.id === selectedId)) closeThread();
    syncThreads(visible);
    renderThread();
    if (data.revision && frameRevision && data.revision !== frameRevision) {
      byId('review-state').textContent = 'A newer document version is available. Your draft is kept.';
      if (!byId('reload')) {
        const reload = button('Load latest version', () => { saveDraft(); location.reload(); }); reload.id = 'reload'; byId('review-state').after(reload);
      }
      byId('post').disabled = true;
    }
  }
  function syncThreads(roots = data.comments.filter(comment => comment.parent_id === null && (byId('resolved').checked || comment.status === 'open'))) {
    tell({ type: 'review-threads', revision: frameRevision, selectedId, threads: roots.filter(comment => comment.revision === frameRevision).map(comment => ({
      id: comment.id, name: comment.name, body: comment.body, quote: comment.quote, selector: comment.selector,
      initial: [...comment.name.trim()][0]?.toUpperCase() || '?', color: avatarColor(comment.name), markup: safeMarkup(comment.markup),
    })) });
  }
  function placeThread(position = threadPosition) {
    threadPosition = position;
    const r = frame.getBoundingClientRect(), width = threadPopover.offsetWidth || 240;
    const right = Math.min(innerWidth, r.right || innerWidth) - width - 12;
    const beside = r.left + (position?.x ?? 24);
    const left = beside > right ? beside - width - 48 : beside;
    threadPopover.style.left = Math.max(12, Math.min(right, left)) + 'px';
    threadPopover.style.top = Math.max(64, Math.min(innerHeight - threadPopover.offsetHeight - 100, r.top + (position?.y ?? 80))) + 'px';
  }
  function closeThread(restoreFocus = true) {
    if (!selectedId && threadPopover.hidden) return;
    selectedId = null; threadPopover.hidden = true; threadPosition = null;
    for (const row of document.querySelectorAll('[data-thread]')) row.setAttribute('aria-expanded', 'false');
    syncThreads();
    if (restoreFocus) threadReturnFocus?.focus({ preventScroll: true });
  }
  function openThread(id) {
    const comment = data.comments.find(item => item.id === id && item.parent_id === null);
    if (!comment || busy) return;
    if (!popover.hidden) closeComposer();
    threadReturnFocus = document.activeElement;
    selectedId = id; threadPosition = null; setTool('read');
    if (matchMedia('(max-width:760px)').matches) closePanel();
    for (const row of document.querySelectorAll('[data-thread]')) row.setAttribute('aria-expanded', String(row.dataset.thread === id));
    renderThread(); threadPopover.hidden = false; placeThread(); syncThreads();
    tell({ type: 'review-focus-thread', revision: frameRevision, id });
    byId('thread-close').focus({ preventScroll: true });
  }
  function renderThread() {
    const comment = data.comments.find(item => item.id === selectedId);
    if (!comment) return;
    const replies = data.comments.filter(item => item.parent_id === comment.id);
    const key = JSON.stringify([comment, replies, data.open, frameRevision, Boolean(viewing)]);
    if (key === threadContentKey) return;
    threadContentKey = key;
    const content = byId('thread-content'); content.replaceChildren();
    for (const item of [comment, ...replies]) {
      const row = element('div', '', 'thread-message');
      row.append(commentHead(item.name, new Date(item.created_at).toLocaleDateString(undefined, { month: 'short', day: 'numeric' }), ''), element('p', item.body, 'comment-body'));
      content.append(row);
    }
    if (comment.quote) content.append(element('blockquote', comment.quote));
    const actions = byId('thread-actions'); actions.replaceChildren();
    if (data.open && !viewing) actions.append(button('Reply', () => selectTarget({ quote: comment.quote, selector: comment.selector }, comment.id, threadPosition, null)));
    if (comment.revision !== frameRevision) {
      content.append(element('p', 'Earlier version', 'hint'));
      actions.append(button('View original version', () => viewMarkup(comment)));
    } else if (comment.markup && !viewing) {
      const details = element('details', '', 'original-layout');
      details.append(element('summary', 'Original layout'), button('View saved layout', () => viewMarkup(comment)));
      actions.append(details);
    }
    placeThread();
  }
  byId('thread-close').addEventListener('click', () => closeThread());
  addEventListener('resize', () => { if (!threadPopover.hidden) placeThread(); });
  byId('stage').addEventListener('scroll', () => { if (!threadPopover.hidden) placeThread(); }, { passive: true });
  async function refresh() {
    if (loading) return; loading = true;
    try {
      const response = await fetch(endpoint, { cache: 'no-store' });
      const result = await response.json();
      if (!response.ok) throw new Error(result.error || 'Could not load comments.');
      const changedAccess = data.open !== result.open;
      data = result; if (changedAccess) setTool(tool); render();
    } catch (error) { byId('review-state').textContent = error.message + ' Use Refresh to retry.'; }
    finally { loading = false; }
  }
  byId('show').addEventListener('click', () => panel.hidden ? openPanel(true) : closePanel());
  byId('close').addEventListener('click', closePanel);
  function activateShortcut(key) {
    if (busy || viewing || !popover.hidden) return false;
    const button = [...document.querySelectorAll('[data-shortcut]')].find(button => button.dataset.shortcut === key);
    if (!button || button.disabled) return false;
    button.click(); return true;
  }
  addEventListener('keydown', event => { const key = reviewShortcut(event); if (key && activateShortcut(key)) event.preventDefault(); });
  for (const button of document.querySelectorAll('[data-tool]')) button.addEventListener('click', () => {
    returnFocus = button;
    if (['undo', 'redo'].includes(button.dataset.tool)) { tell({ type: 'review-history', action: button.dataset.tool }); return; }
    closeThread(false); if (!popover.hidden) closeComposer(); if (!panel.hidden) closePanel(); setTool(button.dataset.tool);
  });
  byId('cancel-compose').addEventListener('click', closeComposer);
  byId('draft').addEventListener('click', () => { targetLabel(); openComposer(); });
  byId('new').addEventListener('click', () => selectTarget({ quote: '', selector: '' }, null, undefined, null));
  byId('clear-target').addEventListener('click', () => { if (confirm('Remove the selected target and marks? Your comment text stays.')) { clearMarks(); anchor = { quote: '', selector: '' }; parentId = null; draftRevision = frameRevision; targetLabel(); saveDraft(); render(); } });
  byId('refresh').addEventListener('click', refresh);
  byId('save').addEventListener('click', () => void refresh());
  byId('resolved').addEventListener('change', render);
  form.addEventListener('input', () => { requestId = newRequestId(); saveDraft(); });
  addEventListener('keydown', event => { if (event.key === 'Escape') { closeThread(); setTool('read'); if (!popover.hidden) closeComposer(); else if (!panel.hidden) closePanel(); } if ((event.metaKey || event.ctrlKey) && event.key === 'Enter' && !popover.hidden) { event.preventDefault(); form.requestSubmit(); } });
  frame.addEventListener('load', () => frame.contentWindow?.postMessage({ type: 'review-ping' }, '*'));
  addEventListener('message', event => {
    if (event.source !== frame.contentWindow || !event.data || typeof event.data !== 'object') return;
    const message = event.data;
    if (message.revision === frameRevision && message.type === 'review-open-thread' && typeof message.id === 'string') openThread(message.id);
    if (message.revision === frameRevision && message.type === 'review-thread-position' && message.id === selectedId && message.position && Number.isFinite(message.position.x) && Number.isFinite(message.position.y)) placeThread(message.position);
    if (message.type === 'review-shortcut' && message.revision === frameRevision && typeof message.key === 'string') activateShortcut(message.key);
    if (message.type === 'review-ready' && typeof message.revision === 'string' && /^[a-f0-9]{64}$/.test(message.revision)) {
      frameRevision = message.revision;
      if (typeof message.title === 'string' && message.title.trim()) { byId('title').textContent = message.title.slice(0, 200); document.title = message.title.slice(0, 200) + ' · Review'; }
      if (viewing) { if (message.revision !== viewing.revision) return; loadingSnapshot = false; tell({ type: 'review-restore', ...viewing, markup: null, replay: true }); render(); tell({ type: 'review-focus-thread', revision: frameRevision, id: viewing.id }); return; }
      if (draftRevision && draftRevision !== frameRevision && (body.value.trim() || markup)) {
        status.textContent = 'Earlier-version draft kept. Clear its selection or choose a new target before posting.';
        sizeFrame(null); targetLabel(); setTool('read'); render(); return;
      }
      draftRevision = frameRevision; sizeFrame(markup); tell({ type: 'review-restore', markup, ...anchor, revision: frameRevision }); setTool(tool); saveDraft();
      document.querySelector('[data-tool="undo"]').disabled = !markup;
      render();
    }
    if (message.type === 'review-markup' && !viewing && !busy && message.revision === frameRevision) {
      const next = safeMarkup(message.markup); if (message.markup && !next) return;
      if (next === null && markup === null) return;
      if (typeof message.quote !== 'string' || typeof message.selector !== 'string') return;
      const nextAnchor = { quote: message.quote.slice(0, 1000), selector: message.selector.slice(0, 500) };
      if (body.value.trim() && (parentId || anchor.selector !== nextAnchor.selector || anchor.quote !== nextAnchor.quote) && !confirm('Change the target for your current draft?')) {
        tell({ type: 'review-restore', markup, ...anchor, revision: frameRevision }); return;
      }
      anchor = nextAnchor; parentId = null;
      markup = next; requestId = newRequestId(); sizeFrame(markup); saveDraft(); targetLabel();
      document.querySelector('[data-tool="undo"]').disabled = !message.canUndo;
      document.querySelector('[data-tool="redo"]').disabled = !message.canRedo;
    }
    if (message.type === 'review-target' && data.open && !viewing && message.revision === frameRevision && typeof message.quote === 'string' && typeof message.selector === 'string') {
      const next = safeMarkup(message.markup); if (message.markup && !next) return;
      selectTarget({ quote: message.quote.slice(0, 1000), selector: message.selector.slice(0, 500) }, null, message.position, next);
    }
    if (message.type === 'review-notice' && typeof message.text === 'string') byId('instruction').textContent = message.text.slice(0, 200);
    if (message.type === 'review-escape') { closeThread(); setTool('read'); if (!popover.hidden) closeComposer(); else if (!panel.hidden) closePanel(); }
  });
  form.addEventListener('submit', async event => {
    event.preventDefault(); if (busy || !data.open || viewing || (!parentId && (data.revision !== frameRevision || draftRevision !== frameRevision))) return;
    busy = true; setTool('read'); render(); status.textContent = 'Posting…'; saveDraft();
    try {
      const response = await fetch(endpoint, { method: 'POST', headers: { 'content-type': 'application/json' }, body: JSON.stringify({ id: requestId, name: name.value, body: body.value, revision: frameRevision, ...anchor, markup, parentId }) });
      const result = await response.json();
      if (!response.ok) throw new Error(result.error || 'Could not post.');
      body.value = ''; clearMarks(); anchor = { quote: '', selector: '' }; parentId = null; requestId = newRequestId(); saveDraft(); status.textContent = 'Posted.';
      closeComposer(); byId('instruction').textContent = 'Comment saved.';
      await refresh();
    } catch (error) { status.textContent = 'Not sent. ' + error.message; }
    finally { busy = false; setTool('read'); render(); byId('post').textContent = body.value ? 'Retry' : 'Post'; }
  });
  async function viewMarkup(comment) {
    if (loadingSnapshot || busy) return;
    const next = safeMarkup(comment.markup);
    saveDraft(); closeThread(false); setTool('read'); popover.hidden = true; if (!panel.hidden) closePanel(); loadingSnapshot = true;
    try {
      const response = await fetch('/' + slug + '/revision/' + comment.revision, { method: 'HEAD' });
      if (!response.ok) throw new Error('Saved document unavailable. The comment and geometry remain in the feedback JSON.');
      viewing = { ...comment, markup: next }; sizeFrame(next); byId('current').hidden = false; byId('snapshot-note').hidden = false;
      byId('draft').hidden = true; frameRevision = ''; frame.src = '/' + slug + '/revision/' + comment.revision; render();
    } catch (error) { byId('instruction').textContent = error.message; loadingSnapshot = false; }
  }
  byId('current').addEventListener('click', () => { closeThread(false); viewing = null; loadingSnapshot = false; sizeFrame(markup); byId('current').hidden = true; byId('snapshot-note').hidden = true; frameRevision = ''; frame.src = '/' + slug + '/content'; render(); });
  function keyboardLayout() {
    if (!visualViewport || !matchMedia('(max-width:760px)').matches) { panel.style.bottom = ''; panel.style.maxHeight = ''; popover.style.bottom = ''; popover.style.maxHeight = ''; return; }
    panel.style.bottom = Math.max(0, innerHeight - visualViewport.height - visualViewport.offsetTop) + 'px';
    panel.style.maxHeight = Math.max(160, visualViewport.height - 56) + 'px';
    popover.style.bottom = Math.max(0, innerHeight - visualViewport.height - visualViewport.offsetTop) + 'px';
    popover.style.maxHeight = Math.max(160, visualViewport.height - 64) + 'px';
  }
  visualViewport?.addEventListener('resize', () => {
    keyboardLayout();
    requestAnimationFrame(() => {
      const focused = document.activeElement;
      if (focused === name || focused === body) popover.scrollTop += focused.getBoundingClientRect().top - popover.getBoundingClientRect().top - 70;
    });
  });
  form.addEventListener('focusin', () => requestAnimationFrame(keyboardLayout));
  visualViewport?.addEventListener('scroll', keyboardLayout);
  targetLabel();
  if (!matchMedia('(max-width:760px)').matches) openPanel();
  frame.contentWindow?.postMessage({ type: 'review-ping' }, '*');
  if (body.value) openComposer();
  void refresh();
  setInterval(() => { if (!document.hidden) void refresh(); }, 15_000);
})();
